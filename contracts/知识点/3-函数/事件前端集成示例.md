## 使用 ethers.js 监听事件

### 1. 基础事件监听

```javascript
import { ethers } from 'ethers';

// 连接到提供商
const provider = new ethers.providers.Web3Provider(window.ethereum);
const signer = provider.getSigner();

// 合约实例
const contractAddress = "0x...";
const contractABI = [...];
const contract = new ethers.Contract(contractAddress, contractABI, signer);

// 监听 Transfer 事件
contract.on("Transfer", (from, to, amount, event) => {
    console.log("Transfer event detected:");
    console.log(`From: ${from}`);
    console.log(`To: ${to}`);
    console.log(`Amount: ${ethers.utils.formatEther(amount)}`);
    console.log("Event:", event);
});
```

### 2. 监听一次性事件

```javascript
// 只监听一次
contract.once("NFTMinted", (tokenId, creator, metadata, royaltyPercent, timestamp) => {
    console.log("NFT minted:", {
        tokenId: tokenId.toString(),
        creator,
        metadata,
        royaltyPercent: royaltyPercent.toNumber(),
        timestamp: new Date(timestamp.toNumber() * 1000)
    });
});
```

### 3. 使用过滤器监听特定事件

```javascript
// 创建过滤器：只监听特定地址的转账
const userAddress = await signer.getAddress();
const filter = contract.filters.Transfer(userAddress, null);  // from = userAddress, to = any

contract.on(filter, (from, to, amount, event) => {
    console.log(`You sent ${ethers.utils.formatEther(amount)} to ${to}`);
});

// 监听收到的转账
const receiveFilter = contract.filters.Transfer(null, userAddress);  // from = any, to = userAddress

contract.on(receiveFilter, (from, to, amount, event) => {
    console.log(`You received ${ethers.utils.formatEther(amount)} from ${from}`);
});
```

### 4. 查询历史事件

```javascript

    "Transfer",
    -10000,  // 从当前区块往前 10000 个区块
    "latest"
);

// 使用过滤器查询
const userTransferFilter = contract.filters.Transfer(userAddress, null);
const userTransfers = await contract.queryFilter(userTransferFilter, 0, "latest");
```

### 5. 移除事件监听器

```javascript
// 定义事件处理函数
const handleTransfer = (from, to, amount, event) => {
    console.log(`Transfer: ${from} -> ${to}, amount: ${amount}`);
};

// 添加监听器
contract.on("Transfer", handleTransfer);

// 移除特定监听器
contract.off("Transfer", handleTransfer);

// 移除所有 Transfer 事件的监听器
contract.removeAllListeners("Transfer");

// 移除所有事件的监听器

```

## 使用 web3.js 监听事件

### 1. 基础事件监听

```javascript
import Web3 from 'web3';

const web3 = new Web3(window.ethereum);
const contractAddress = "0x...";
const contractABI = [...];
const contract = new web3.eth.Contract(contractABI, contractAddress);

// 监听事件
contract.events.Transfer({
    fromBlock: 'latest'
})
.on('data', (event) => {
    console.log('Transfer event:', event.returnValues);
})
.on('error', (error) => {
    console.error('Error:', error);
});
```

### 2. 使用过滤器

```javascript
// 监听特定地址的转账
const userAddress = "0x...";

contract.events.Transfer({
    filter: { from: userAddress },  // 过滤 indexed 参数
    fromBlock: 0
})
.on('data', (event) => {
    console.log('Transfer from user:', event.returnValues);
})
.on('changed', (event) => {
    // 移除事件（发生链重组）
    console.log('Event removed:', event);
})
.on('error', console.error);
```

### 3. 查询历史事件

```javascript
// 获取过去的事件
contract.getPastEvents('Transfer', {
    filter: { from: userAddress },
    fromBlock: 0,
    toBlock: 'latest'

.then((events) => {

        console.log({
            from: event.returnValues.from,
            to: event.returnValues.to,
            amount: web3.utils.fromWei(event.returnValues.amount, 'ether'),
            blockNumber: event.blockNumber,
            transactionHash: event.transactionHash
        });
    });
})
.catch((error) => {
    console.error('Error fetching events:', error);
});
```

### 4. 监听所有事件

```javascript
// 监听合约的所有事件
contract.events.allEvents({
    fromBlock: 'latest'
})
.on('data', (event) => {
```

### 2. 时间范围查询

```javascript
// 查询特定时间范围内的事件
async function getEventsByTimeRange(startTime, endTime) {
    // 获取起始时间对应的区块号
    const startBlock = await getBlockByTimestamp(startTime);
    const endBlock = await getBlockByTimestamp(endTime);
    
    // 查询事件
    const events = await contract.queryFilter("Transfer", startBlock, endBlock);
    return events;
}

// 辅助函数：根据时间戳查找区块号（二分查找）
async function getBlockByTimestamp(timestamp) {
    const latestBlock = await provider.getBlockNumber();
    const latestBlockData = await provider.getBlock(latestBlock);
    
    if (latestBlockData.timestamp <= timestamp) {
        return latestBlock;
    }
    
    let low = 0;
    let high = latestBlock;
    
    while (low < high) {
        const mid = Math.floor((low + high) / 2);
        const block = await provider.getBlock(mid);
        
        if (block.timestamp < timestamp) {
            low = mid + 1;
        } else {
            high = mid;
        }
    }
    
    return low;
}
```

### 3. 分页查询大量事件

```javascript
// 分批查询事件，避免一次性查询过多数据
async function queryEventsInBatches(eventName, fromBlock, toBlock, batchSize = 5000) {
    const events = [];
    
    for (let start = fromBlock; start <= toBlock; start += batchSize) {
        const end = Math.min(start + batchSize - 1, toBlock);
        console.log(`Querying blocks ${start} to ${end}...`);
        
        const batchEvents = await contract.queryFilter(eventName, start, end);
        events.push(...batchEvents);
    }
    
    return events;
}

// 使用示例
const allTransfers = await queryEventsInBatches("Transfer", 0, "latest");
console.log(`Total events found: ${allTransfers.length}`);
```

## 事件数据处理

### 1. 解析和格式化事件数据

```javascript
// 格式化事件数据
function formatTransferEvent(event) {
    return {
        // 基本信息
        from: event.args.from,
        to: event.args.to,
        amount: ethers.utils.formatEther(event.args.amount),
        
        // 区块信息
        blockNumber: event.blockNumber,
        blockHash: event.blockHash,
        transactionHash: event.transactionHash,
        
        // 事件索引
        logIndex: event.logIndex,
        transactionIndex: event.transactionIndex,
        
        // 时间戳（需要额外查询）
        timestamp: null  // 稍后填充
    };
}

// 添加时间戳
async function addTimestamp(eventData) {
    const block = await provider.getBlock(eventData.blockNumber);
    eventData.timestamp = new Date(block.timestamp * 1000);
    return eventData;
}

// 使用示例
contract.on("Transfer", async (from, to, amount, event) => {
    const formattedData = formatTransferEvent(event);
    const dataWithTimestamp = await addTimestamp(formattedData);
    console.log(dataWithTimestamp);
});
```

### 2. 构建交易历史

```javascript
// 获取用户的完整交易历史
async function getUserTransactionHistory(userAddress) {
    // 获取发送的转账
    const sentFilter = contract.filters.Transfer(userAddress, null);
    const sentEvents = await contract.queryFilter(sentFilter, 0, "latest");
    
    // 获取接收的转账
    const receivedFilter = contract.filters.Transfer(null, userAddress);
    const receivedEvents = await contract.queryFilter(receivedFilter, 0, "latest");
    
    // 合并并排序
    const allEvents = [...sentEvents, ...receivedEvents];
    allEvents.sort((a, b) => {
        if (a.blockNumber !== b.blockNumber) {
            return a.blockNumber - b.blockNumber;
        }
        return a.logIndex - b.logIndex;
    });
    
    // 格式化
    const history = await Promise.all(
        allEvents.map(async (event) => {
            const formattedData = formatTransferEvent(event);
            const dataWithTimestamp = await addTimestamp(formattedData);
            
            // 标记交易类型
            dataWithTimestamp.type = event.args.from === userAddress ? 'sent' : 'received';
            
            return dataWithTimestamp;
        })
    );
    
    return history;
}

// 使用示例
const history = await getUserTransactionHistory("0x...");
console.log(`Total transactions: ${history.length}`);
history.forEach(tx => {
    console.log(`${tx.type}: ${tx.amount} at ${tx.timestamp}`);
});
```

### 3. 事件数据缓存

```javascript
// 使用 localStorage 缓存事件数据
class EventCache {
    constructor(contractAddress, eventName) {
        this.cacheKey = `events_${contractAddress}_${eventName}`;
        this.lastBlockKey = `lastBlock_${contractAddress}_${eventName}`;
    }
    
    // 获取缓存的事件
    getCachedEvents() {
        const cached = localStorage.getItem(this.cacheKey);
        return cached ? JSON.parse(cached) : [];
    }
    
    // 获取最后查询的区块号
    getLastBlock() {
        const lastBlock = localStorage.getItem(this.lastBlockKey);
        return lastBlock ? parseInt(lastBlock) : 0;
    }
    
    // 保存事件到缓存
    saveEvents(events, currentBlock) {
        const cached = this.getCachedEvents();
        const allEvents = [...cached, ...events];
        
        localStorage.setItem(this.cacheKey, JSON.stringify(allEvents));
        localStorage.setItem(this.lastBlockKey, currentBlock.toString());
    }
    
    // 清除缓存
    clearCache() {
        localStorage.removeItem(this.cacheKey);
        localStorage.removeItem(this.lastBlockKey);
    }
}

// 使用缓存查询事件
async function queryEventsWithCache(contract, eventName) {
    const cache = new EventCache(contract.address, eventName);
    
    // 获取缓存的事件
    const cachedEvents = cache.getCachedEvents();
    const lastBlock = cache.getLastBlock();
    
    // 只查询新的事件
    const currentBlock = await provider.getBlockNumber();
    const newEvents = await contract.queryFilter(
        eventName,
        lastBlock + 1,
        currentBlock
    );
    
    // 保存到缓存
    if (newEvents.length > 0) {
        cache.saveEvents(newEvents, currentBlock);
    }
    
    // 返回所有事件
    return [...cachedEvents, ...newEvents];
}
```

### 1. 使用 React Hooks 监听事件

```jsx
import { useState, useEffect } from 'react';
import { ethers } from 'ethers';

// 自定义 Hook：监听合约事件
function useContractEvent(contract, eventName, filter = null) {
    const [events, setEvents] = useState([]);
    
    useEffect(() => {
        if (!contract) return;
        
        // 事件处理函数
        const handleEvent = (...args) => {
            const event = args[args.length - 1];  // 最后一个参数是事件对象
            
            setEvents(prevEvents => [...prevEvents, {
                args: event.args,
                blockNumber: event.blockNumber,
                transactionHash: event.transactionHash
            }]);
        };
        
        // 添加监听器
        if (filter) {
            contract.on(filter, handleEvent);
        } else {
            contract.on(eventName, handleEvent);
        }
        
        // 清理函数
        return () => {
            if (filter) {
                contract.off(filter, handleEvent);
            } else {
                contract.off(eventName, handleEvent);
            }
        };
    }, [contract, eventName, filter]);
    
    return events;
}

// 使用示例
function TransferMonitor({ contract, userAddress }) {
```

```jsx
import { useState, useEffect } from 'react';

function NFTMarketplace({ contract }) {
    const [nfts, setNfts] = useState([]);
    const [recentSales, setRecentSales] = useState([]);
    
    useEffect(() => {
        if (!contract) return;

                tokenId: tokenId.toString(),
                creator,
                owner: creator,

                royaltyPercent: royaltyPercent.toNumber(),

                    ? { ...nft, forSale: true, price: ethers.utils.formatEther(price) }
                    : nft
            ));
        };
        
        // 监听 NFT 出售
        const handleSale = (tokenId, seller, buyer, price, platformFee, royaltyFee, timestamp) => {
            // 更新 NFT 列表
            setNfts(prev => prev.map(nft => 
                nft.tokenId === tokenId.toString()
                    ? { ...nft, owner: buyer, forSale: false }
                    : nft

        contract.on("NFTMinted", handleMint);
        contract.on("NFTListed", handleListing);

        // 清理

            contract.off("NFTMinted", handleMint);
            contract.off("NFTListed", handleListing);

            })).slice(0, 10);
            
            setRecentSales(sales);
        }
        
        loadHistoricalData();
    }, [contract]);
    
    return (

            <h1>NFT Marketplace</h1>
            
            <section>
                <h2>All NFTs ({nfts.length})</h2>
                <div className="nft-grid">
```

### 3. 实时通知组件

```jsx
import { useState, useEffect } from 'react';
import { ethers } from 'ethers';

function EventNotifications({ contract, userAddress }) {
    const [notifications, setNotifications] = useState([]);
    
    useEffect(() => {
        if (!contract || !userAddress) return;
        
        // 监听与用户相关的事件
        const handleUserEvent = (eventName, message) => {
            const notification = {
                id: Date.now(),
                type: eventName,
                message,
                timestamp: new Date()
            };
            
            setNotifications(prev => [notification, ...prev]);
            
            // 5 秒后自动移除
            setTimeout(() => {
                setNotifications(prev => prev.filter(n => n.id !== notification.id));
            }, 5000);
        };
        
        // 接收转账
        const receiveFilter = contract.filters.Transfer(null, userAddress);
        contract.on(receiveFilter, (from, to, amount) => {
            handleUserEvent(
                'Transfer',
                `You received ${ethers.utils.formatEther(amount)} ETH from ${from.substring(0, 8)}...`
            );
        });
        
        // NFT 购买成功
        const buyFilter = contract.filters.NFTSold(null, userAddress);
        contract.on(buyFilter, (tokenId, seller, buyer, price) => {
            handleUserEvent(
                'Purchase',
                `You successfully purchased NFT #${tokenId} for ${ethers.utils.formatEther(price)} ETH`
            );
        });
        
        return () => {
            contract.removeAllListeners(receiveFilter);
            contract.removeAllListeners(buyFilter);
        };
    }, [contract, userAddress]);
    
    return (
        <div className="notifications">
            {notifications.map(notification => (
                <div key={notification.id} className={`notification ${notification.type}`}>
                    <p>{notification.message}</p>
                    <small>{notification.timestamp.toLocaleTimeString()}</small>
                </div>
            ))}
        </div>
    );
}
```

## 总结

### 事件监听最佳实践

1. **性能优化**
   - 使用过滤器减少不必要的事件处理
   - 分批查询大量历史事件
   - 缓存已查询的事件数据
   - 及时移除不需要的监听器

2. **错误处理**
   - 始终添加错误处理回调
   - 处理网络断开重连
   - 处理链重组（reorg）情况

3. **用户体验**
   - 提供实时反馈
   - 显示加载状态
   - 合理使用通知
   - 缓存数据减少等待时间

4. **安全考虑**
   - 验证事件数据的合法性
   - 不要完全依赖事件数据做关键判断
   - 注意事件可能被重组或回滚

5. **代码组织**
   - 使用自定义 Hook 封装事件监听逻辑
   - 集中管理事件监听器
   - 确保组件卸载时清理监听器
